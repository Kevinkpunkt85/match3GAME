        }

      function swapTilesState(r1, c1, r2, c2) {
        let temp = board[r1][c1];
        board[r1][c1] = board[r2][c2];
        board[r2][c2] = temp;

      function checkMatches() {
        let matchedSet = new Set();
        // Horizontal
        for (let r = 0; r < ROWS; r++) {
          for (let c = 0; c < COLS - 2; c++) {
            let val = board[r][c];
            if (val !== null && val === board[r][c + 1] && val === board[r][c + 2]) {
              matchedSet.add(`${r},${c}`);
              matchedSet.add(`${r},${c+1}`);
              matchedSet.add(`${r},${c+2}`);
            }
          }
        }
        // Vertical
        for (let c = 0; c < COLS; c++) {
          for (let r = 0; r < ROWS - 2; r++) {
            let val = board[r][c];
            if (val !== null && val === board[r + 1][c] && val === board[r + 2][c]) {
              matchedSet.add(`${r},${c}`);
              matchedSet.add(`${r+1},${c}`);
              matchedSet.add(`${r+2},${c}`);
            }
          }
        }
        let matches = [];
        matchedSet.forEach(item => {
          let [r, c] = item.split(',').map(Number);
          matches.push({
            r,
            c
          });
        });
        return matches;
      }
      async function processMatchesAndCascade() {
        isProcessing = true;
        let matches = checkMatches();
        let chainCount = 0;
        while (matches.length > 0) {
          chainCount++;
          score += matches.length * 10 * chainCount;
          updateScoreDisplay();
          playSound('match');
          for (let m of matches) {
            const el = document.querySelector(`[data-row='${m.r}'][data-col='${m.c}']`);
            if (el) el.classList.add('tile-match');
          }
          await new Promise(res => setTimeout(res, 250));
          for (let m of matches) {
            board[m.r][m.c] = null;
          }
          for (let c = 0; c < COLS; c++) {
            let emptyRow = ROWS - 1;
            for (let r = ROWS - 1; r >= 0; r--) {
              if (board[r][c] !== null) {
                board[emptyRow][c] = board[r][c];
                if (emptyRow !== r) {
                  board[r][c] = null;
                }
                emptyRow--;
              }
            }
            for (let r = emptyRow; r >= 0; r--) {
              board[r][c] = Math.floor(Math.random() * TILES.length);
            }
          }
          renderBoard();
          await new Promise(res => setTimeout(res, 200));
          matches = checkMatches();
        }
        isProcessing = false;
      }

      function shuffleBoard() {
        if (!isPlaying || isProcessing) return;
        isProcessing = true;
        let tilesList = [];
        for (let r = 0; r < ROWS; r++) {
          for (let c = 0; c < COLS; c++) {
            tilesList.push(board[r][c]);
          }
        }
        let valid = false;
        let attempts = 0;
        while (!valid && attempts < 50) {
          attempts++;
          for (let i = tilesList.length - 1; i > 0; i--) {
            let j = Math.floor(Math.random() * (i + 1));
            [tilesList[i], tilesList[j]] = [tilesList[j], tilesList[i]];
          }
          let idx = 0;
          for (let r = 0; r < ROWS; r++) {
            for (let c = 0; c < COLS; c++) {
              board[r][c] = tilesList[idx++];
            }
          }
          if (checkMatches().length === 0) {
            valid = true;
          }
        }
        renderBoard();
        isProcessing = false;
        playSound('swap');
      }

        document.getElementById('score-display').innerText = String(score).padStart(5, '0');
